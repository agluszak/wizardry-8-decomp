cmake_minimum_required(VERSION 3.20)
project(wiz8_fid_seeds LANGUAGES C CXX)

set(FID_ZLIB_SOURCE "" CACHE PATH "Pinned zlib source tree")
set(FID_JPEG_SOURCE "" CACHE PATH "Pinned IJG JPEG source tree")
set(FID_INFOZIP_SOURCE "" CACHE PATH "Pinned Info-ZIP source tree")

function(add_seed TARGET SOURCE_ROOT)
    set(multi SOURCES OPTIONS DEFINITIONS INCLUDES)
    cmake_parse_arguments(SEED "" "" "${multi}" ${ARGN})
    set(sources ${SEED_SOURCES})
    list(TRANSFORM sources PREPEND "${SOURCE_ROOT}/")
    add_library(${TARGET} OBJECT EXCLUDE_FROM_ALL ${sources})
    target_compile_options(${TARGET} PRIVATE ${SEED_OPTIONS})
    target_compile_definitions(${TARGET} PRIVATE ${SEED_DEFINITIONS})
    target_include_directories(${TARGET} PRIVATE ${SEED_INCLUDES})
    add_dependencies(fid-seeds ${TARGET})
endfunction()

add_custom_target(fid-seeds)
set(seed_count 0)
if(FID_ZLIB_SOURCE)
    file(GLOB units CONFIGURE_DEPENDS RELATIVE "${FID_ZLIB_SOURCE}" "${FID_ZLIB_SOURCE}/*.c")
    list(REMOVE_ITEM units example.c minigzip.c)
    list(SORT units)
    foreach(variant release_md_o2 release_md_o2_g6 release_md_o1)
        set(options /nologo /MD /O2)
        if(variant STREQUAL release_md_o1)
            set(options /nologo /MD /O1)
        elseif(variant STREQUAL release_md_o2_g6)
            list(APPEND options /G6)
        endif()
        add_seed(fid_zlib_1_0_4_${variant} "${FID_ZLIB_SOURCE}" SOURCES ${units}
            OPTIONS ${options} INCLUDES "${FID_ZLIB_SOURCE}")
        math(EXPR seed_count "${seed_count} + 1")
    endforeach()
endif()
if(FID_JPEG_SOURCE)
    file(GLOB units CONFIGURE_DEPENDS RELATIVE "${FID_JPEG_SOURCE}" "${FID_JPEG_SOURCE}/j*.c")
    list(REMOVE_ITEM units jpegtran.c jmemansi.c jmemdos.c jmemname.c)
    list(SORT units)
    foreach(variant release_md_o2 release_md_o1)
        set(optimization /O2)
        if(variant STREQUAL release_md_o1)
            set(optimization /O1)
        endif()
        add_seed(fid_ijg_jpeg_6_${variant} "${FID_JPEG_SOURCE}" SOURCES ${units}
            OPTIONS /nologo /MD ${optimization}
            INCLUDES "${CMAKE_CURRENT_LIST_DIR}/../../config/fid-overlays/ijg-jpeg-6" "${FID_JPEG_SOURCE}")
        math(EXPR seed_count "${seed_count} + 1")
    endforeach()
endif()
if(FID_INFOZIP_SOURCE)
    file(GLOB units CONFIGURE_DEPENDS RELATIVE "${FID_INFOZIP_SOURCE}"
        "${FID_INFOZIP_SOURCE}/*.c" "${FID_INFOZIP_SOURCE}/win32/*.c"
        "${FID_INFOZIP_SOURCE}/windll/*.c")
    list(REMOVE_ITEM units apihelp.c envargs.c funzip.c unzip.c unzipstb.c
        win32/crc_i386.c windll/example.c)
    list(SORT units)
    foreach(variant upstream_release_mt_o2 target_runtime_md_o2)
        set(runtime /MD)
        if(variant STREQUAL upstream_release_mt_o2)
            set(runtime /MT)
        endif()
        add_seed(fid_infozip_unzip_5_4_${variant} "${FID_INFOZIP_SOURCE}" SOURCES ${units}
            OPTIONS /nologo /Zp4 ${runtime} /W3 /GX /O2
            DEFINITIONS NDEBUG WIN32 _WINDOWS WINDLL USE_EF_UT_TIME DLL
            INCLUDES "${FID_INFOZIP_SOURCE}" "${FID_INFOZIP_SOURCE}/windll" "${FID_INFOZIP_SOURCE}/win32")
        math(EXPR seed_count "${seed_count} + 1")
    endforeach()
endif()
if(seed_count EQUAL 0)
    message(FATAL_ERROR "configure with at least one FID source path")
endif()
